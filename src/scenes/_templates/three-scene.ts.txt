// ---------------------------------------------------------------------------
// THREE.JS custom scene template.
//
// Prereq:  npm install three
// Use it:  copy this file to  src/scenes/06-three-cube/scene.ts
//
// A CustomScene is handed its own <canvas> and returns a { dispose() } handle.
// The raw-WebGL engine does nothing here — Three.js owns the canvas. This is
// the whole point of the escape hatch: your GLSL knowledge transfers to
// ShaderMaterial, but the plumbing is Three's.
// ---------------------------------------------------------------------------
import * as THREE from 'three';
import type { CustomScene, SceneInstance } from '../../engine/types';

const scene: CustomScene = {
  kind: 'custom',
  id: '06-three-cube',
  title: 'Three: shader cube',
  level: 3,
  blurb: 'Same GLSL, wrapped in a Three.js ShaderMaterial on a real mesh.',

  mount(canvas: HTMLCanvasElement): SceneInstance {
    const renderer = new THREE.WebGLRenderer({ canvas, antialias: true });
    const setSize = () => {
      const w = canvas.clientWidth, h = canvas.clientHeight;
      renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
      renderer.setSize(w, h, false);
      camera.aspect = w / h;
      camera.updateProjectionMatrix();
    };

    const three = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(50, 1, 0.1, 100);
    camera.position.set(0, 0, 3);

    const material = new THREE.ShaderMaterial({
      uniforms: { uTime: { value: 0 } },
      vertexShader: /* glsl */ `
        varying vec2 vUv;
        void main() {
          vUv = uv;
          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
        }`,
      fragmentShader: /* glsl */ `
        varying vec2 vUv;
        uniform float uTime;
        void main() {
          vec3 col = 0.5 + 0.5 * cos(uTime + vUv.xyx + vec3(0.0, 2.0, 4.0));
          gl_FragColor = vec4(col, 1.0);
        }`,
    });

    const mesh = new THREE.Mesh(new THREE.BoxGeometry(1.2, 1.2, 1.2), material);
    three.add(mesh);

    const clock = new THREE.Clock();
    let raf = 0;
    const loop = () => {
      raf = requestAnimationFrame(loop);
      const t = clock.getElapsedTime();
      material.uniforms.uTime.value = t;
      mesh.rotation.x = t * 0.5;
      mesh.rotation.y = t * 0.7;
      renderer.render(three, camera);
    };

    const ro = new ResizeObserver(setSize);
    ro.observe(canvas);
    setSize();
    loop();

    return {
      dispose() {
        cancelAnimationFrame(raf);
        ro.disconnect();
        mesh.geometry.dispose();
        material.dispose();
        renderer.dispose();
      },
    };
  },
};

export default scene;
